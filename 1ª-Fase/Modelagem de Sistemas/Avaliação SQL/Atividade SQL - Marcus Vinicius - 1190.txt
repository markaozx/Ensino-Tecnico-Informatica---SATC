create database escritorio de engenharia

create table setores(
codigo		int(5)		not null,
nome		varchar(50)	not null,
gerente		varchar(50)	not null,
telefone	int(20)		not null,
primary key (codigo));

create table engenheiros (
cpf		int(15)		not null,
nome		varchar(50)	not null,
cargo		varchar(50)	not null,
salario		float(20,2)	not null,
telefone	int(20)		not null,
endereço	varchar(50)	not null,
cidade		varchar(50)	not null,
estado		varchar(50)	not null,
codsetor	int(5)		not null,
primary key (telefone)
foreign key (codsetor) references setores (codigo));

create table projetos (
codigo		int(5)		not null,
descrição	varchar(300)	not null,
inicio		varchar(25)	not null,
fim		varchar(25)	not null,
tipo		varchar(50)	not null,
valor		float(20,2)	not null,
codsetor	int(5)		not null,
primary key (codigo)
foreign (codsetor) references setores (codigo));

create table clientes (
nome		varchar(50)	not null,
cpf		int(15)		not null,
telefone	int(20)		not null,
endereço	varchar(50)	not null,
cidade		varchar(50)	not null,
estado		varchar(50)	not null,
codprojeto	int(5)		not null,
primary key (nome)
foreign (codprojeto) references projetos (codigo));
