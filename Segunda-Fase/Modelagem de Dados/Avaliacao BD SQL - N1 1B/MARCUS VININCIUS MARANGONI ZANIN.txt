create database petshop;

create table clientes (
cpf			int(11)			not null,
nome		varchar(50) 		not null,
endereco	varchar(25)		not null,
bairro		varchar(15)		not null,
cidade		varchar(15)		not null,
estado		varchar(15)		not null,
telefone		int(15)			not null,
codigo		int(6)			not null,
primary key (codigo));

create table funcionarios (
cpf			int(11)			not null,
nome		varchar(50) 		not null,
endereco	varchar(25)		not null,
bairro		varchar(15)		not null,
cidade		varchar(15)		not null,
estado		varchar(15)		not null,
telefone		int(15)			not null,
cargo		varchar(20)		not null,
salario		float(6,2)			not null,
codigo		int(6)			not null,
primary key (codigo));

create table animal (
nome		varchar(20)		not null,
raça 		varchar(20)		not null,
especie		varchar(20)		not null,
peso		float(4,2)		        not null,
altura		int(4)			not null,
cor			varchar(15)		not null,
idade		int(3)			not null,
codigo		int(6)			not null,
codcliente	int(6)			not null,
primary key (codigo),
foreign key (codcliente) references cliente(codigo));

create table atendimentos (
codigo		int(6)			not null,
data		date			not null,
hora		time				not null,
servico		varchar(20)		not null,
valor		float(6,2)			not null,
codfunci	int(6)			not null,
codanimal	int(6)			not null,
codcliente	int(6)			not null,
primary key (codigo),
foreign key (codfunci) references funcionario(codigo),
foreign key (codanimal) references animal(codigo),
foreign key (codcliente) references cliente(codigo));
